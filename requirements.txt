# =============================================================================
# AI 代码导师系统 —— 运行期依赖清单
#
# 用途：start.bat 会自动执行 `pip install -r requirements.txt` 安装以下依赖，
#       学生双击一个 .bat 就能把整套系统跑起来，不需要懂 pip。
#
# 覆盖的功能模块（详见 README.md）：
#   · 代码文件自动分类      app/services/file_classifier.py
#   · AI 自动检测（语法+深度）app/services/code_checker.py
#   · AI 代码改错           app/services/code_fixer.py
#   · AI 注释生成           app/services/comment_generator.py
#   · 开源协作链路（检索/测试生成/沙箱/自动修复）app/agents/*、app/services/repo/*
#
# 注意：本文件必须与 pyproject.toml 的 [project.dependencies] 保持一致，
#       否则 `pip install -r requirements.txt` 与 `pip install -e .` 装出来的环境会不同。
#       可以用 `python tools/verify_integration.py` 自动比对两者。
# =============================================================================

pydantic-settings>=2.3.0

python-multipart>=0.0.9

# ---------- 数据库（本地 SQLite，异步驱动）----------

# ---------- 代码解析：Git 操作 + 多语言语法树 ----------
GitPython>=3.1.43

# 内网 / 离线环境会直接下载超时失败。

tree-sitter-javascript>=0.23

# ---------- 代码检索：关键词召回 + 向量召回 ----------

# fastembed 走 ONNX Runtime 推理，避免 sentence-transformers 带来的约 2GB PyTorch 依赖。
# 首次使用会下载嵌入模型（约 180MB），公网 huggingface.co 不可达时
# 已默认走镜像（见 .env.example 的 RETRIEVAL__HF_ENDPOINT）。
fastembed>=0.4

# ---------- 中文分词（检索用）----------
jieba>=0.42

# ---------- 沙箱执行：容器调用 + 覆盖率采集 ----------

# coverage：本地沙箱后端跑测试时要用的覆盖率工具，属于运行期依赖而非纯开发工具
docker>=7.1
coverage>=7.5.0
